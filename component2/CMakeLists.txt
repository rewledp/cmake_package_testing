set(COMPONENT_FUCK_SHEEP "fuck_sheep")
add_library(${COMPONENT_FUCK_SHEEP} src/fuck_sheep.cpp)

target_compile_features(${COMPONENT_FUCK_SHEEP} PUBLIC cxx_std_20)

target_include_directories(${COMPONENT_FUCK_SHEEP} PUBLIC
	$<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/${CMAKE_INSTALL_INCLUDEDIR}>
	$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

add_library(${PROJECT_NAME}::${COMPONENT_FUCK_SHEEP} ALIAS ${COMPONENT_FUCK_SHEEP})

include(GNUInstallDirs)

install(
	DIRECTORY include/
	DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${PROJECT_NAME}
)

install(TARGETS ${COMPONENT_FUCK_SHEEP}
	EXPORT "${COMPONENT_FUCK_SHEEP}_export"
	RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
	LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
	ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
)

install(EXPORT "${COMPONENT_FUCK_SHEEP}_export"
	FILE "${COMPONENT_FUCK_SHEEP}_export.cmake"
	NAMESPACE "${PROJECT_NAME}::"
	DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/${PROJECT_NAME}"
)
